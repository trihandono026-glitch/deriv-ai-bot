async function connectMarket(){ marketSymbol = "cryBTCUSD"; publicWS = new WebSocket(PUBLIC_WS); bindSocket(publicWS); await new Promise((resolve,reject)=>{ const timer=setTimeout( ()=>reject(Error("Koneksi data pasar timeout.")),12000 ); publicWS.onopen=()=>{ clearTimeout(timer); resolve(); }; publicWS.onerror=()=>{ clearTimeout(timer); reject(Error("Gagal terhubung ke data pasar.")); }; }); // Uji apakah simbol benar-benar tersedia const test = await wsRequest(publicWS,{ ticks_history: marketSymbol, count: 2, end: "latest", style: "ticks" }); if(!test.history && !test.candles){ throw Error("Deriv tidak mengirim data untuk cryBTCUSD."); } await wsRequest(publicWS,{ ticks: marketSymbol, subscribe: 1 }); log("BTCUSD berhasil ditemukan: " + marketSymbol); }
